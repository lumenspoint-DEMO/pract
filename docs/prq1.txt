def get_documents():
    n = int(input("Enter the number of documents: "))
    documents = {}
    for i in range(n):
        doc = input(f"Enter Document {i + 1}: ").lower().split()
        documents[i + 1] = doc
    return documents
def build_index(documents):
    term_docs = []

    for doc_id, terms in documents.items():
        for term in terms:
            term_docs.append((term, doc_id))

    print("\nSTEP 1: Term and Document ID")
    for term, doc_id in term_docs:
        print(term, "\t", doc_id)

    term_docs.sort()

    print("\nSTEP 2: Arranged Term and Document ID")
    for term, doc_id in term_docs:
        print(term, "\t", doc_id)

    posting_lists = {}
    for term, doc_id in term_docs:
        posting_lists.setdefault(term, set()).add(doc_id)

    print("\nSTEP 3: Term and Document Frequency")
    for term in sorted(posting_lists):
        print(f"{term}\t{len(posting_lists[term])}")

    print("\nSTEP 4: Posting Lists")
    for term in sorted(posting_lists):
        print(f"{term} -> {sorted(posting_lists[term])}")

    return posting_lists


documents = get_documents()
build_index(documents)

def accept_docs():
    num_of_docs = int(input("Enter no. of docs: "))
    docs = []

    for i in range(1, num_of_docs + 1):
        doc = input(f"Enter the content of doc{i}: ").lower().split()
        docs.append(doc)

    print("ALL Documents:", docs)

    full_doc = []
    for doc in docs:
        full_doc.extend(doc)

    full_terms = list(dict.fromkeys(full_doc))
    return docs, full_terms


def print_incident_matrix(docs, full_terms):
    matrix = {}

    for term in full_terms:
        vector = [1 if term in doc else 0 for doc in docs]
        matrix[term] = vector

    return matrix


def query_terms(matrix, query):
    results = []
    i = 0

    while i < len(query):
        word = query[i]

        if word == "not" and i + 1 < len(query):
            term = query[i + 1]
            if term in matrix:
                vector = [1 if v == 0 else 0 for v in matrix[term]]
                results.append(vector)
            i += 2
        else:
            if word in matrix:
                results.append(matrix[word])
            i += 1

    if not results:
        return [0] * len(next(iter(matrix.values())))

    final = results[0]
    for vector in results[1:]:
        final = [a & b for a, b in zip(final, vector)]

    return final

import math
from collections import Counter

n = int(input("Enter number of documents: "))
documents = []

for i in range(n):
    documents.append(input("Enter Document " + str(i+1) + ": ").lower())

query = input("Enter Query: ").lower()

print("\nSTEP 1 : Documents and Query")
for i in range(n):
    print("Document", i+1, ":", documents[i])
print("Query :", query)

print("\nSTEP 2 : Frequency Table")
for i in range(n):
    print("\nDocument", i+1)
    for word, count in Counter(documents[i].split()).items():
        print(word, ":", count)

vocabulary = []
for doc in documents:
    for word in doc.split():
        if word not in vocabulary:
            vocabulary.append(word)

vocabulary.sort()

print("\nSTEP 3 : Vocabulary")
print(vocabulary)

print("\nSTEP 4 : Arranged Vocabulary")
print(vocabulary)

print("\nSTEP 5 : Vectors")
vectors = []

for i in range(n):
    v = []
    for word in vocabulary:
        v.append(1 if word in documents[i].split() else 0)
    vectors.append(v)
    print("Document", i+1, "Vector :", v)

q = [1 if word in query.split() else 0 for word in vocabulary]
print("Query Vector :", q)

print("\nSTEP 6 : Cosine Similarity")
similarity = []

for v in vectors:
    dot = sum(v[i] * q[i] for i in range(len(q)))
    a = math.sqrt(sum(x*x for x in v))
    b = math.sqrt(sum(x*x for x in q))
    
    if a == 0 or b == 0: 
        sim = 0
    else:
        sim = dot / (a * b)
    
    similarity.append(sim)

for i in range(n):
    print("Document", i+1, "=", round(similarity[i], 4))

print("\nSTEP 7 : Ranked Documents")
rank = sorted(enumerate(similarity, 1), key=lambda x: x[1], reverse=True)

for doc, score in rank:
    print("Document", doc, ":", round(score, 4))

str1 = input("Enter the first String: ")

def edit_distance(str1, str2):
    n = len(str1)
    m = len(str2)

    dp = [[0 for _ in range(n + 1)] for _ in range(m + 1)]

    for i in range(m + 1):
        dp[i][0] = i

    for j in range(n + 1):
        dp[0][j] = j

    for i in range(1, m + 1):
        for j in range(1, n + 1):
            if str2[i - 1] == str1[j - 1]:
                dp[i][j] = dp[i - 1][j - 1]
            else:
                dp[i][j] = 1 + min(
                    dp[i - 1][j],
                    dp[i][j - 1],
                    dp[i - 1][j - 1]
                )

    return dp


dp = edit_distance(str1, str2)

print("\nEdit Distance Matrix:")
for row in dp:
    print(row)

print("\nFinal Edit Distance =", dp[len(str2)][len(str1)])

    rows = [row for row in data if row[2] == fever]

    print("P(Fever) =", round(prior, 4))

print("P(N | Flu,Covid):", round(score["N"], 4))

    print("Predicted Fever: Y")

    return math.sqrt((a[0]-b[0])**2 + (a[1]-b[1])**2)

#Web Crawling and Indexing
import requests
from bs4 import BeautifulSoup
from urllib.parse import urljoin
from urllib.robotparser import RobotFileParser

seen = set()

def crawl(url, depth=2):
    if url in seen or depth == 0:
        return
    seen.add(url)

    r = RobotFileParser(urljoin(url, "/robots.txt"))
    r.read()

    if not r.can_fetch("MiniCrawler", url):
        print("Blocked:", url)
        return

    soup = BeautifulSoup(requests.get(url).text, "html.parser")
    print("Crawled:", url)

    for a in soup.find_all("a", href=True):
        link = urljoin(url, a["href"])
        if link.startswith("http"):
            crawl(link, depth - 1)

crawl("https://.vercel.app", 2)

#Implement the PageRank algorithm

import numpy as np

n = int(input("Enter number of nodes: "))
links = int(input("Enter the number of links: "))

adj_matrix = np.zeros((n, n))

print("Enter links (from_node to_node):")

for _ in range(links):
    a, b = map(int, input().split())
    a -= 1
    b -= 1
    adj_matrix[b][a] = 1

print("\nAdjacency Matrix:")
print(adj_matrix)

M = np.zeros((n, n))

for j in range(n):
    outgoing = sum(adj_matrix[:, j])

    if outgoing != 0:
        for i in range(n):
            M[i][j] = adj_matrix[i][j] / outgoing
    else:
        # Dangling node: distribute equally to all nodes.
        M[:, j] = 1 / n

print("\nTransition Matrix M:")
print(np.round(M, 4))

iterations = int(input("Enter number of iterations: "))

PR = np.ones(n) / n

print("\nPageRank Vector for each iteration:")

for i in range(iterations):
    PR = np.dot(M, PR)
    print(f"Iteration {i + 1}: {np.round(PR, 4)}")

highest = np.argmax(PR)

print("\nNode with highest PageRank:")
print(
    "Node",
    highest + 1,
    "with PageRank value",
    round(PR[highest], 4)
)

#Apply HITS Algorithm to a small web graph and analyze the results.
def create_graph(nodes, links):
    graph = {node: [] for node in nodes}

    for u, v in links:
        if u not in graph or v not in graph:
            raise ValueError("Link contains an unknown node.")
        graph[u].append(v)

    return graph


def display_graph(graph):
    print("\nDirected Graph:")
    for node in graph:
        print(node, "->", graph[node])

def normalize(scores):
    norm = sum(value * value for value in scores.values()) ** 0.5

    if norm == 0:
        return {node: 0 for node in scores}

    return {node: scores[node] / norm for node in scores}


def find_best(scores):
    node = max(scores, key=scores.get)
    return node, scores[node]


def hits():
    nodes = input("Enter nodes separated by space: ").split()

    number_of_links = int(input("Enter number of links: "))
    links = []

    for i in range(number_of_links):
        u, v = input(f"Link {i + 1} (from to): ").split()
        links.append((u, v))

    iterations = int(input("Enter number of iterations: "))

    graph = create_graph(nodes, links)
    display_graph(graph)

    hub = {node: 1 for node in nodes}
    authority = {node: 1 for node in nodes}

    for iteration in range(1, iterations + 1):
        authority = {
            node: sum(
                hub[source]
                for source in graph
                if node in graph[source]
            )
            for node in nodes
        }

        authority = normalize(authority)

        hub = {
            node: sum(authority[target] for target in graph[node])
            for node in nodes
        }

        hub = normalize(hub)

        print(f"\nIteration: {iteration}")
        print("\nNode\tAuthority\tHub")

        for node in nodes:
            print(
                f"{node}\t"
                f"{authority[node]:.4f}\t\t"
                f"{hub[node]:.4f}"
            )

    best_authority, authority_score = find_best(authority)
    best_hub, hub_score = find_best(hub)

    print("\n========================================")
    print("Final Result")
    print("========================================")
    print(
        "Best Authority Node:",
        best_authority,
        "Score:",
        round(authority_score, 4)
    )
    print(
        "Best Hub Node:",
        best_hub,
        "Score:",
        round(hub_score, 4)
    )


if __name__ == "__main__":
    hits()

#Program to Rank Documents using SVM (Support Vector Machine)
import numpy as np
from sklearn.svm import SVC
X = np.array([
    [2, 5],
    [4, 3],
    [6, 7],
    [5, 2],
    [7, 6],
    [3, 4]
])

y = np.array([0, 0, 1, 1, 1, 0])
pairs = []
labels = []

for i in range(len(X)):
    for j in range(len(X)):
        if y[i] > y[j]:
            pairs.append(X[i] - X[j])
            labels.append(1)
            pairs.append(X[j] - X[i])
            labels.append(0)
pairs = np.array(pairs)
labels = np.array(labels)
model = SVC(kernel="linear")
model.fit(pairs, labels)
weights = model.coef_[0]
scores = X @ weights
rank = np.argsort(scores)[::-1]

print("Ranked Documents:")

for i in rank:
    print(f"D{i + 1} Score = {scores[i]:.2f}")

import nltk
from nltk.tokenize import sent_tokenize, word_tokenize
from collections import Counter

text = """Artificial Intelligence is changing the world.
Machine learning is an important part of Artificial Intelligence.
AI is used in healthcare and education.
AI helps computers learn from data."""

sentences = sent_tokenize(text)

words = word_tokenize(text.lower())
freq = Counter(
    word for word in words if word.isalnum()
)

scores = {}

for sentence in sentences:
    sentence_words = word_tokenize(sentence.lower())
    scores[sentence] = sum(
        freq[word] for word in sentence_words if word.isalnum()
    )

summary = sorted(
    sentences,
    key=lambda sentence: scores[sentence],
    reverse=True
)[:2]

print("Summary:")
print(" ".join(summary))
