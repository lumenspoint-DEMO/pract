#1) Perform testing of hypothesis for mean of one sample using z-test.

import numpy as np
from scipy.stats import norm

n = int(input("Enter number of values: "))

data = []
for i in range(n):
    data.append(float(input(f"Enter value {i+1}: ")))

mu = float(input("Enter population mean: "))
sigma = float(input("Enter population standard deviation: "))
alpha = float(input("Enter significance level: "))

z = (np.mean(data) - mu) / (sigma / np.sqrt(n))
p = norm.cdf(z)

print("\nSample Mean =", np.mean(data))
print("Z-value =", z)
print("P-value =", p)

if p < alpha:
    print("H0 is rejected")
else:
    print("H0 is accepted")
#INPUT
# Enter number of values: 5
# Enter value 1: 48
# Enter value 2: 52
# Enter value 3: 50
# Enter value 4: 49
# Enter value 5: 51
# Enter population mean: 50
# Enter population standard deviation: 2
# Enter significance level: 0.05

#3) Perform testing of hypothesis for equality of mean of two samples using z-test.

import numpy as np
from scipy.stats import norm

n1 = int(input("Enter number of values in sample 1: "))

data1 = []
for i in range(n1):
    data1.append(float(input(f"Enter sample 1 value {i+1}: ")))

n2 = int(input("Enter number of values in sample 2: "))

data2 = []
for i in range(n2):
    data2.append(float(input(f"Enter sample 2 value {i+1}: ")))

sigma1 = float(input("Enter standard deviation of sample 1: "))
sigma2 = float(input("Enter standard deviation of sample 2: "))
alpha = float(input("Enter significance level: "))

z = (np.mean(data1) - np.mean(data2)) / np.sqrt(
    sigma1**2 / n1 + sigma2**2 / n2
)

p = 2 * norm.sf(abs(z))

print("\nMean 1 =", np.mean(data1))
print("Mean 2 =", np.mean(data2))
print("Z-value =", z)
print("P-value =", p)

if p < alpha:
    print("H0 is rejected")
else:
    print("H0 is accepted")

#1) Perform testing of hypothesis for variance of one sample.

import numpy as np
from scipy.stats import chi2

n = int(input("Enter number of values: "))

data = []
for i in range(n):
    data.append(float(input(f"Enter value {i+1}: ")))

variance = float(input("Enter population variance: "))
alpha = float(input("Enter significance level: "))

s2 = np.var(data, ddof=1)

chi = (n - 1) * s2 / variance

p = 2 * min(chi2.cdf(chi, n - 1), chi2.sf(chi, n - 1))

print("\nSample Variance =", s2)
print("Chi-square value =", chi)
print("P-value =", p)

if p < alpha:
    print("H0 is rejected")
else:
    print("H0 is accepted")
#INPUT
# Enter number of values: 5
# Enter value 1: 48
# Enter value 2: 52
# Enter value 3: 50
# Enter value 4: 49
# Enter value 5: 51
# Enter population variance: 2
# Enter significance level: 0.05


#2) Perform testing of hypothesis for variance of two samples.

import numpy as np
from scipy.stats import f

n1 = int(input("Enter number of values in sample 1: "))

data1 = []
for i in range(n1):
    data1.append(float(input(f"Enter sample 1 value {i+1}: ")))

n2 = int(input("Enter number of values in sample 2: "))

data2 = []
for i in range(n2):
    data2.append(float(input(f"Enter sample 2 value {i+1}: ")))

alpha = float(input("Enter significance level: "))

s1 = np.var(data1, ddof=1)
s2 = np.var(data2, ddof=1)

F = s1 / s2
p = f.cdf(F, n1 - 1, n2 - 1)

print("\nVariance 1 =", s1)
print("Variance 2 =", s2)
print("F-value =", F)
print("P-value =", p)

if p < alpha:
    print("H0 is rejected")
else:
    print("H0 is accepted")

#2) Perform testing of hypothesis using two-way ANOVA (WOR).

import pandas as pd
import statsmodels.api as sm
from statsmodels.formula.api import ols

rows = int(input("Enter number of rows: "))
cols = int(input("Enter number of columns: "))

data = []

for i in range(rows):
    row = []
    print(f"\n--- Row {i+1} ---")

    for j in range(cols):
        row.append(float(input(f"Enter value {j+1}: ")))

    data.append(row)

table = pd.DataFrame(data).stack().reset_index()
table.columns = ["Row", "Column", "Value"]

model = ols("Value ~ C(Row) + C(Column)", data=table).fit()
result = sm.stats.anova_lm(model, typ=2)

alpha = float(input("\nEnter significance level: "))

print("\n===== TWO-WAY ANOVA (WOR) =====")

for name, p in zip(["Row", "Column"], result["PR(>F)"]):
    print(f"\n{name}")
    print("P-value :", round(p, 4))

    if p < alpha:
        print("Decision : H0 is rejected")
    else:
        print("Decision : H0 is accepted")

#3) Perform testing of hypothesis using two-way ANOVA (WR).

import pandas as pd
import statsmodels.api as sm
from statsmodels.formula.api import ols

rows = int(input("Enter number of rows: "))
cols = int(input("Enter number of columns: "))
n = int(input("Enter values per cell: "))

row_data = []
col_data = []
values = []

for i in range(rows):
    for j in range(cols):
        print(f"\n--- Row {i+1}, Column {j+1} ---")

        for k in range(n):
            value = float(input(f"Enter value {k+1}: "))

            row_data.append(i)
            col_data.append(j)
            values.append(value)

table = pd.DataFrame({
    "Row": row_data,
    "Column": col_data,
    "Value": values
})

model = ols("Value ~ C(Row) * C(Column)", data=table).fit()
result = sm.stats.anova_lm(model, typ=2)

alpha = float(input("\nEnter significance level: "))

print("\n===== TWO-WAY ANOVA (WR) =====")

names = ["Row", "Column", "Interaction"]

for name, p in zip(names, result["PR(>F)"][:3]):
    print(f"\n{name}")
    print("P-value :", round(p, 4))

    if p < alpha:
        print("Decision : H0 is rejected")
    else:
        print("Decision : H0 is accepted")

negative = sum(x < median for x in data)

if p < alpha:
