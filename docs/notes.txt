from collections import deque
def water_jug_bfs():
    max_a, max_b = 6, 5
    goal = 3
    start = (0, 0)
    queue = deque([start])
    visited = {start}
    parent = {start: None}
    
    while queue:
        a, b = queue.popleft()
        if a == goal or b == goal:
            print("Found a Solution!\n")
            # Print solution path
            path = []
            state = (a, b)
            while state is not None:
                path.append(state)
                state = parent[state]
            path.reverse()

            print("Solution Path:")
            for s in path:
                print(f"Jug A: {s[0]}L, Jug B: {s[1]}L")
            return

        # Generate all possible next states
        possible_states = [
            (max_a, b),                                  # Fill Jug A
            (a, max_b),                                  # Fill Jug B
            (0, b),                                      # Empty Jug A
            (a, 0),                                      # Empty Jug B
            (min(a + b, max_a), b - (min(a + b, max_a) - a)),  # Pour B -> A
            (a - (min(a + b, max_b) - b), min(a + b, max_b))   # Pour A -> B
        ]
        for state in possible_states:
            if state not in visited:
                visited.add(state)
                parent[state] = (a, b)
                queue.append(state)
    print("No solution found.")

def bfs(start):
    visited = set()
    queue = deque([(start, [])])
    while queue:
        state,path =queue.popleft()
        if state  == goal:
            return path + [state]
        if state in visited:

            queue.append((''.join(new_state), path + [state]))
    return None

def dfs(graph, root, end):
    visited=[]
    if root not in visited:
        visited.append(root)
        for i in graph[root]:
            if visited[-1] is end:
                break
            dfs(graph, i , visited, end)
    return visited

def solve(board, row, n):
    if row == n:
        for r in board:
            print(" ".join(r))
        print()
        return True

    for col in range(n):
        safe = True

        # Check column
        for i in range(row):
            if board[i][col] == "Q":
                safe = False

        # Check upper-left diagonal
        i, j = row - 1, col - 1
        while i >= 0 and j >= 0:
            if board[i][j] == "Q":
                safe = False
            i -= 1
            j -= 1

        # Check upper-right diagonal
        i, j = row - 1, col + 1
        while i >= 0 and j < n:
            if board[i][j] == "Q":
                safe = False
            i -= 1
            j += 1

        if safe:
            board[row][col] = "Q"
            if solve(board, row + 1, n):
                return True
            board[row][col] = "."
    return False

n = int(input("Enter N: "))
board = [["." for _ in range(n)] for _ in range(n)]
if not solve(board, 0, n):
    print("No solution found")

        'D':[4,3],
        'E':[6,2],
        'F':[2,1],
        'G':[9,5]}
def minmax_alpha_beta(node, depth, alpha, beta, max_player):
    if depth == 0:
        if node in tree:
            return tree[node][0] if max_player else tree[node][0]
        else:
            return node
    if max_player:
        value=float('-inf')
        for child in tree[node]:
            value=max(value,minmax_alpha_beta(child,depth-1,alpha, beta,False))
            alpha=max(alpha,value)
            if alpha>=beta:
                print(f'Pruning branch at node{node}')
                break
        return value
    else:
        value=float('inf')
        for child in tree[node]:
            value=min(value,minmax_alpha_beta(child,depth-1,alpha, beta, True))
            beta=min(beta,value)
            if alpha>=beta:
                print(f'Pruning branch at node{node}')
                break
        return value

Q.12)cannibal and missionaries
from collections import deque

def bfs():
    start = (3, 3, 0)
    goal = (0, 0, 1)

    moves = [(1,0), (2,0), (0,1), (0,2), (1,1)]

    def valid(m, c):
        if m < 0 or c < 0 or m > 3 or c > 3:
            return False

        if m > 0 and c > m:
            return False

        mr, cr = 3-m, 3-c

        if mr > 0 and cr > mr:
            return False

        return True

    queue = deque([(start, [])])
    visited = set()

    while queue:
        state, path = queue.popleft()

        if state in visited:
            continue

        visited.add(state)
        path = path + [state]

        if state == goal:
            return path

        m, c, boat = state

        for dm, dc in moves:
            if boat == 0:
                new = (m-dm, c-dc, 1)
            else:
                new = (m+dm, c+dc, 0)

            if valid(new[0], new[1]):
                queue.append((new, path))


solution = bfs()

for i, state in enumerate(solution):
    print("Step", i, ":", state)
