# # Q1. Water Jug – State Space Possibilities

from collections import deque

def water_jug(a, b, goal):
    q = deque([((0,0), [])])
    seen = set()

    while q:
        state, path = q.popleft()

        if state in seen:
            continue

        seen.add(state)
        path = path + [state]

        if state == goal:
            return path

        x, y = state

        next_states = [
            (a, y),
            (x, b),
            (0, y),
            (x, 0),
            (max(0, x - (b - y)), min(b, x + y)),
            (min(a, x + y), max(0, y - (a - x)))
        ]

        for s in next_states:
            if s not in seen:
            q.append((s, path))

    return None


solution = water_jug(5, 4, (2, 0))

print("Solution Path:")
print(solution)

def bfs(start):

    visited = set()

    while q:
        state, path = q.popleft()

        if state == goal:
            return path + [state]

        if state in visited:

        zero = state.index("0")

            s = list(state)
            s[zero], s[m] = s[m], s[zero]
            new = "".join(s)

            if new not in visited:
                q.append((new, path + [state]))

def hanoi(n, source, helper, destination):
    if n == 1:
        print("Move disk 1 from", source, "to", destination)
        return

    hanoi(n-1, source, destination, helper)

    print("Move disk", n, "from", source, "to", destination)

    hanoi(n-1, helper, source, destination)

from collections import deque

def bfs(graph, start):
    visited = []
    queue = deque([start])

    while queue:
        node = queue.popleft()

        if node not in visited:
            visited.append(node)
            queue.extend(graph[node])

    return visited


def all_paths(graph, start, end, path=[]):
    path = path + [start]

    if start == end:
        print(path)
        return

    for node in graph[start]:
        if node not in path:
            all_paths(graph, node, end, path)


def shortest_path(graph, start, end):
    queue = deque([[start]])

    while queue:
        path = queue.popleft()
        node = path[-1]

        if node == end:
            return path

        for next_node in graph[node]:
            if next_node not in path:
                queue.append(path + [next_node])

def dfs(g, n, seen, d):
    if n not in seen:
        seen.append(n)

        for i in g[n]:
            if seen[-1] == d:
                break

            dfs(g, i, seen, d)

        return seen

def greedy(start, goal):
    current = start
    path = [current]
    while current != goal:
        neighbours = graph[current]
        if not neighbours:
            return []

        next_node = None
        smallest = 999

        for node in neighbours:
            if h[node] < smallest:
                smallest = h[node]
                next_node = node

        print("Choosing:", next_node)

        current = next_node
        path.append(current)
    return path

def astar(start, goal):
    path = [start]
    g = 0

    while start != goal:
        best = None
        f = 999

        for node in graph[start]:
            new_f = g + graph[start][node] + h[node]

            if new_f < f:
                f = new_f
                best = node

        g += graph[start][best]
        start = best
        path.append(best)

    return path

# N-Queen Problem

def solve(board, row, n):
    if row == n:
        for r in board:
            print(" ".join(r))
        return True

    for col in range(n):
        safe = True

        for r in range(row):
            if board[r][col] == "Q":
                safe = False

        for r, c in zip(range(row-1, -1, -1), range(col-1, -1, -1)):
            if board[r][c] == "Q":
                safe = False

        for r, c in zip(range(row-1, -1, -1), range(col+1, n)):
            if board[r][c] == "Q":
                safe = False

        if safe:
            board[row][col] = "Q"

            if solve(board, row+1, n):
                return True

            board[row][col] = "."

    return False

n = int(input("Enter N: "))

board = [["." for _ in range(n)] for _ in range(n)]

solve(board, 0, n)

    print(f'| {board[0]} | {board[1]} | {board[2]}')
    print('-----------------')

    print(f'| {board[3]} | {board[4]} | {board[5]}')
    print('-----------------')

    print('-----------------')

def is_winner(p):

    return (
        (board[0] == p and board[1] == p and board[2] == p) or
        (board[3] == p and board[4] == p and board[5] == p) or
        (board[6] == p and board[7] == p and board[8] == p) or

        (board[0] == p and board[3] == p and board[6] == p) or
        (board[1] == p and board[4] == p and board[7] == p) or
        (board[2] == p and board[5] == p and board[8] == p) or

        (board[0] == p and board[4] == p and board[8] == p) or
        (board[2] == p and board[4] == p and board[6] == p)
    )


def is_tie():

    return ' ' not in board


def game():
    global player

    while True:
        show_board()

        try:
            move = int(input(f"Player {player}, enter 0-8: "))

            if 0 <= move <= 8 and board[move] == ' ':
                board[move] = player

                if is_winner(player):
                    show_board()
                    print(f"Player {player} wins!")
                    break
                elif is_tie():
                    show_board()
                    print("It's a tie!")
                    break

                if player == 'X':
                    player = 'O'
                else:
                    player = 'X'
            else:
                print("Invalid move. Choose an empty position from 0-8.")

        except (ValueError, IndexError):
            print("Invalid input. Enter a number from 0-8.")

#Q2 Minimax with Alpha-Beta
def minimax(node, depth, alpha, beta, max_player):

    if depth == 0:
        return node

    if max_player:
        value = -999

        for child in tree[node]:
            value = max(value, minimax(child, depth-1,
                                       alpha, beta, False))

            alpha = max(alpha, value)

            if alpha >= beta:
                break

        return value

    else:
        value = 999

        for child in tree[node]:
            value = min(value, minimax(child, depth-1,
                                       alpha, beta, True))

            beta = min(beta, value)

            if alpha >= beta:
                break

        return value

from collections import deque

def is_valid(m_left, c_left, m_total, c_total):
    m_right = m_total - m_left
    c_right = c_total - c_left

    if m_left < 0 or c_left < 0 or m_right < 0 or c_right < 0:
        return False

    if m_left > 0 and c_left > m_left:
        return False

    if m_right > 0 and c_right > m_right:
        return False

    return True

def solve(m_total, c_total, direction):
    boat = 0 if direction.lower() == "left" else 1

    start = (m_total, c_total, boat)
    goal = (0, 0, 1)

    queue = deque([(start, [])])
    visited = set()

    moves = [
        (2, 0),
        (1, 0),
        (0, 2),
        (0, 1),
        (1, 1)
    ]

    while queue:
        state, path = queue.popleft()

        if state in visited:
            continue

        visited.add(state)

        if state == goal:
            return path

        m_left, c_left, boat = state

        for m, c in moves:
            if m + c == 0:
                continue

            if boat == 0:
                new_state = (m_left - m, c_left - c, 1)
                text = f"Move {m} missionary(s) and {c} cannibal(s) Left -> Right"
            else:
                new_state = (m_left + m, c_left + c, 0)
                text = f"Move {m} missionary(s) and {c} cannibal(s) Right -> Left"

            if is_valid(new_state[0], new_state[1], m_total, c_total):
                queue.append((new_state, path + [text]))

    return None


missionaries = int(input("Enter number of missionaries: "))
cannibals = int(input("Enter number of cannibals: "))
direction = input("Enter boat starting direction (left/right): ")

answer = solve(missionaries, cannibals, direction)

if answer:
    print("\nCorrect Solution:\n")

    for i, step in enumerate(answer, 1):
        print(f"{i}. {step}")
else:
    print("No solution exists.")
